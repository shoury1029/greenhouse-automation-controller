/*>>> getADCSample: ===========================================================
Author:		Varun Shoury Golluru 
Date:		14/06/2024
Modified:	None
Desc:		This function will get the ADC sample from ADC module and returns 
			ADC result. 
Input: 		a char type, chen. This input is required to select a channel for 
			ADCON0.
Returns:	an int type. A 10 bit ADC is returned.  
 ============================================================================*/
int getADCSample(char chan)
{
	ADCON0bits.CHS = chan;
	ADCON0bits.GO = TRUE;
	while(ADCON0bits.GO);   
	return ADRES;

}// eo getADCSample ::

/*>>> initSensorCh: ===========================================================
Author:		Varun Shoury 
Date:		01/03/2025
Modified:	None
Desc:		initializes the sensorsCh type data structure with given values.
Input: 		sensorCh_t *senPtr, this input is a pointer, it helps to goto the
			locations of the data structure of sensorCh_t type.
			char chID, this input is a variable which helps in going through
			the switch cases.  
Returns:	None. 
 ============================================================================*/
void initSensorCh(sensorCh_t *senPtr, char chID)
{
	int index = 0,initLimitIndex = 0;
	for(index =0; index<SAMPSIZE; index++)
	{
		senPtr -> samples[index] = 0;
	}
	senPtr -> avg = 0; 
	switch(chID)
	{
		case 0:
			senPtr -> upperLimit = 35;
			senPtr -> lowerLimit = 15;
			break;
		case 1: 
			senPtr -> upperLimit = 65;
			senPtr -> lowerLimit = 35;
			break;
		case 2:
			senPtr -> upperLimit = 1400;
			senPtr -> lowerLimit = 650;
			break;
		default:
			break;
	}
	senPtr -> insert = 0;
	senPtr -> avgRdy = 0;
} // eo initSensorCh::

/*>>> calcCheckSum: ===========================================================
Author:		Varun Shoury 
Date:		25/03/2025
Modified:	None
Desc:		This function is created to perform an Ex-or operation on a string as
			a result it will give an unique value. 
Input: 		char ptr, this is needed to go through an array/string. 
Returns:	char, the return is a unique value from the ex-or operation on the data 
			in the string passed. 
 ==============================================================================*/
char calcCheckSum(char *ptr)
{
	char checkSum = 0;
	while(*ptr)
	{
		checkSum^= *ptr;
		ptr ++;
	}
	return checkSum;
}// eo calcCheckSum::

}// eo changeMode::

/*>>> initTRMT: ===========================================================
Author:		Varun Shoury
Date:		25/03/2025
Modified:	None
Desc:		this function will initialize the comBuf_t type data structure
			members. 
Input: 		comBuf_t *ptr, this input is created to point the address of the 
			member in data structure. 
Returns:	None.
 ============================================================================*/
void initTRMT(comBuf_t *ptr)
{
	int index = 0;
	for(index=0; index<BUFSIZE; index++)
	{
		ptr -> buf[index] = FALSE; 
	}// eo for loop
	ptr -> insertCount = FALSE;
	ptr -> senRdy = FALSE;
}// eo initEvent:: 

/*>>> changeChannel: ===========================================================
Author:		Varun Shoury 
Date:		01/03/2025
Modified:	None
Desc:		This function is used to change the channel, by incrementing the chanSel
			and keeping the variable in limit. 
Input: 		None.
Returns:	None. 
 ============================================================================*/
void changeChannel()
{
	pbs.chanSel ++;
	if(pbs.chanSel >= SENCOUNT) 
	{
		pbs.chanSel = FALSE;
	}
}// eo changeChannel::

/*>>> incrementLimit: ===========================================================
Author:		Varun Shoury 
Date:		01/03/2025
Modified:	None
Desc:		This function is used to increase the limit(upper and lower) using the 
			push buttons. 
Input: 		None.
Returns:	None.
 ============================================================================*/
void incrementLimit()
{
	if(pbs.mode)
	{
		sensors[pbs.chanSel].upperLimit++;
	}
	else
	{
		sensors[pbs.chanSel].lowerLimit++;
	}
}// eo incrementLimit::

/*>>> decrementLimit: ===========================================================
Author:		Varun Shoury 
Date:		01/03/2025
Modified:	None
Desc:		This function is used to decrease the limit(upper and lower) usisng 
			push buttons. 
Input: 		None.
Returns:	None. 
 ============================================================================*/
void decrementLimit()
{
	if(pbs.mode)
	{
		sensors[pbs.chanSel].upperLimit--;
	}
	else
	{
		sensors[pbs.chanSel].lowerLimit--;
	}
}// eo decrementLimit::

void createASentence()
{
	if(pbs.mode)
	{
		// saving the base statemtment(a string) into the array buf[]. 
		sprintf(trmt.buf,"$%s,%d,%d,%d,%d,%d",conlim,CONTROLLER,MYADDY,pbs.chanSel,pbs.mode,sensors[pbs.chanSel].upperLimit);
	}
	else
	{
		sprintf(trmt.buf,"$%s,%d,%d,%d,%d,%d",conlim,CONTROLLER,MYADDY,pbs.chanSel,pbs.mode,sensors[pbs.chanSel].lowerLimit);
	}	
	sprintf(trmt.buf,"%s,%d\r",trmt.buf,calcCheckSum(trmt.buf));

}// eo createASentence::

/*=== MAIN: FUNCTION ==========================================================
 ============================================================================*/
void main( void )
{
	int secCount = 0;	
	char chID = 0;
	sysInitialize();	  //calling all the functions in one function
	initStepper(&vent);
	initPB(&pbs);
	for(chID=0; chID < SENCOUNT; chID++)	 //for loop to initialize all sensor's members to 0
	{
		initSensorCh(&sensors[chID], chID);		// calling initSensorCh function and giving the address of sensors array
	}// eo for loop
	initTRMT(&trmt);
	while(TRUE)								//indefinite loop
	{
		pbs.pbState = PORTA & PBMASK;
		if(pbs.pbState != NOPRESS && pbs.pbState != pbs.lastState)
		{
			pbs.lastState = pbs.pbState;
			switch(pbs.pbState)
			{
				case MODE:
						changeMode();
						break;
				case CHANSEL:
						changeChannel();
						break;
				case INC:
						incrementLimit();
						limitChanged = TRUE;
						trmt.insertCount = 0;
						createASentence();
						//puts2USART(trmt.buf);
						break;
				case DEC:
						decrementLimit();
						limitChanged = TRUE;
						trmt.insertCount = 0;
						createASentence();
						//puts2USART(trmt.buf);
						break;
				default:
						break;
			}// eo switch case
		}// eo if pbstate!=nopress
  		else if(pbs.pbState == NOPRESS)
		{
			pbs.lastState = 0;
		}

		
		if(T0FLAG)							//cheching if timer0 flag bit is true or not
		{
			resetTMR0();					// resetting the timer if flagged 
			secCount ++;
			if(vent.curPos != vent.setPos)
			{
				if(vent.curPos < vent.setPos)
				{
					vent.patCounter ++;                  
					if(vent.patCounter >= PATTERNCOUNT)
					{
						vent.patCounter = FALSE;
					}
					vent.currentPat = patterns[vent.patCounter];
					vent.curPos += STEP;
					STEPPERPORT = STEPPERMASK & vent.currentPat;
				}
				else
				{
					vent.patCounter --;
					if(vent.patCounter < 0)
					{
						vent.patCounter = PATTERNCOUNT - 1;
					}
					vent.currentPat = patterns[vent.patCounter];
					vent.curPos -= STEP;
					STEPPERPORT = STEPPERMASK & vent.currentPat;
				}	
			}
			if(secCount == ONESEC)			//checking if secCount has reached the limit(ONESEC)
			{
				secCount = FALSE;			//making secCount zero
				for(chID =0; chID<SENCOUNT; chID ++)
				{	
					sensors[chID].samples[sensors[chID].insert]= getADCSample(chID);// getting a sample from ADC module for every channel
					sensors[chID].insert ++;	//incrementing insert
				
					if(sensors[chID].insert >= SAMPSIZE)//checking if insert is within the limit
					{
						sensors[chID].insert = FALSE;
						sensors[chID].avgRdy = TRUE;	//making the avgrdy flag bit true if insert reaches the limit
					}// eo if
					if(sensors[chID].avgRdy)			
					{
						long sum = 0;				
						int index = 0;
						for(index=0; index<SAMPSIZE; index++)
						{
							sum+= sensors[chID].samples[index];// finding sum of the samples for each sensor
						}// eo for loop for sum
						sensors[chID].avg = sum/SAMPSIZE;	//finding the average
						switch(chID)
						{
							case 0:
								mTemp = ((sensors[0].avg*VOLCONVERT) - TEMPBVAL)/TEMPMVAL;
								sensors[0].avg = mTemp;
								break;
							case 1:
								mHumd = (sensors[1].avg*VOLCONVERT/HUMDMVAL);
								sensors[1].avg = mHumd;
								break;
							case 2:
								mCO2 = (sensors[2].avg*VOLCONVERT)/CO2MVAL;
								sensors[2].avg = mCO2;
								break;
							default:
								break;
						}// eo switch 
						// checking if temp avg is in limits, when is turn off cooler, heater, fan
						if(sensors[0].avg > sensors[0].lowerLimit && sensors[0].avg < sensors[0].upperLimit)
						{
							COOLER = FALSE;							
							HEATER = FALSE;
							FAN = FALSE;
						}
						/* when temp avg is over upperlimit, cooler and fan are turned on, and position of 
							motor is set to 90degrees. */
						if(sensors[0].avg > sensors[0].upperLimit)
						{
							COOLER = TRUE;
							HEATER = FALSE;
							FAN = TRUE;
							vent.setPos = 90;
						}// eo if 			
						if(sensors[0].avg < sensors[0].lowerLimit)
						{
							COOLER = FALSE;
							HEATER = TRUE;
							FAN = TRUE;
							vent.setPos = 6;
						}// eo if 
						if(sensors[1].avg > sensors[1].lowerLimit && sensors[1].avg < sensors[1].upperLimit)
						{
							SPKLR = FALSE;
						}
						if(sensors[1].avg > sensors[1].upperLimit)
						{
							SPKLR = FALSE;
							vent.setPos = 66;
						}// eo if 			
						if(sensors[1].avg < sensors[1].lowerLimit)
						{
							SPKLR = TRUE;
							vent.setPos = 12;
						}// eo if 
						if(sensors[2].avg > sensors[2].upperLimit)
						{
							FAN = TRUE;
							vent.setPos = 9;
						}// eo if 
						if(sensors[2].avg < sensors[2].lowerLimit)
						{
							FAN = TRUE;
							vent.setPos = 90;
						}// eo if 
					}// eo avg if
				}//eo chID for loop
				if(trmt.senRdy)
				{
					trmt.senRdy = FALSE;
					puts2USART(trmt.buf);
				}
				trmt.insertCount ++;
				display();
			}// eo if secCount is 10
		}//eo flag if 
	}// eo while
} // eo main::

/*>>> calcCheckSum: ===========================================================
Author:		Varun Shoury 
Date:		25/03/2025
Modified:	None
Desc:		This function is created to perform an Ex-or operation on a string as
			a result it will give an unique value. 
Input: 		char ptr, this is needed to go through an array/string. 
Returns:	char, the return is a unique value from the ex-or operation on the data 
			in the string passed. 
 ==============================================================================*/
char calcCheckSum(char *ptr)
{
	char checkSum = 0;
	while(*ptr)
	{
		checkSum ^= *ptr;
		ptr ++;
	}
	return checkSum;
}// eo calcCheckSum::

/*>>> validateSentence: ===========================================================
Author:		Varun Shoury
Date:		14/04/2025
Modified:	None
Desc:		This function will validate the sentence when we feed the sentence to it
			The validation is done by comparing the checksum values of the sent sentence 
			and the new sentence which we got by separating the base sentence. 
Input: 		char *ptr, This is a pointer used to go through the string.
Returns:	char, the data is a boolean value giving out TRUE or FALSE.
=============================================================================*/
char validateSentence(char *ptr)
{
	char newCs = 0;
	char rcvdCs = 0;
	int count = strlen(ptr);
	// isolating rcvd checkSum 
	char csRdy = FALSE;
	while(csRdy != TRUE)
	{
		if(*(ptr+count) == ',')
		{	
			*(ptr+count) = 0;
			rcvdCs = atoi(ptr+count+1);
			csRdy = TRUE;
		}
		count --;
	}// eo while
	newCs = calcCheckSum(ptr);
	if(newCs == rcvdCs)
	{
		return TRUE;
	}
	else
	{
		return FALSE;
	}
}// eo validate Sentence ::

/*>>> parseStr: ===========================================================
Author:		Varun Shoury
Date:		14/04/2025
Modified:	None
Desc:		This function will chop the string into multiple strings so that 
			they can be worked with individually. 
Input: 		char *ptr, it is a pointer which goes through the feeded string. 
Returns:	None
=============================================================================*/

void parseStr(char *ptr)
{
	char tokenCounter = 0;
	while(*ptr)
	{
		if(*ptr == '$' || *ptr == ',')
		{
			*ptr = 0; // forcing a null in the string 
			tokens[tokenCounter] = ptr + 1;
			tokenCounter ++;
		}// eo if 
		ptr ++;
	}// eo while
}// eo parseStr::

/*>>> execStr: ===========================================================
Author:		Varun Shoury
Date:		14/04/2025
Modified:	None
Desc:		This function will perform comparisions in order to reach the upper 
			and lower limits, will compare by using the tokens from parsed 
			string. 
Input: 		None
Returns:	None
 =============================================================================*/
void execStr()
{
	if(strcmp(conlim,tokens[0])==0)
	{
		if(atoi(tokens[4]))

		}
		else
		{
			sensors[atoi(tokens[3])].lowerLimit = atoi(tokens[5]);
		}// eo else 
	}// eo strcmp if
}// eo execStr::

/*>>> isr: ==================================================================
Author:		Varun Shoury Golluru 
Date:		14/04/2025
Modified:	None
Desc:		This fucntion will take the string byte by byte with RCREG2 and 
			will do the service routine. 
Input: 		None 
Returns:	None 
 ============================================================================*/
void isr()
{
	if(RC2FLAG)
	{
		char hold = 0;		  	
		hold = RC2BYTE;
		if(hold == '$')
		{
			rcvr2.insert = 0;	
		}
		if(hold == '\r')
		{
			rcvr2.senRdy = TRUE;
			hold = 0;
		}
		rcvr2.buf[rcvr2.insert] = hold;
		rcvr2.insert ++;
	}
	INTCON |= 0xC0;
}// eo ISR::

	while(TRUE)

			rcvr2.senRdy = FALSE;
			if(validateSentence(rcvr2.buf))
			{
				parseStr(rcvr2.buf);
				execStr();	
			}
		}
		display();

	}		
} // eo main::
